import { useRef } from 'react';

export function useDragResize(onChange) {
  const state = useRef(null);

  const onMouseDownMove = (e, start, cb) => {
    e.stopPropagation();
    const startX = e.clientX, startY = e.clientY;
    const cleanup = () => {
      window.removeEventListener('mousemove', move);
      window.removeEventListener('mouseup', up);
    };
    const move = (ev) => {
      const dx = ev.clientX - startX;
      const dy = ev.clientY - startY;
      cb(dx, dy);
    };
    const up = () => { cleanup(); };
    window.addEventListener('mousemove', move);
    window.addEventListener('mouseup', up);
  };

  const startDrag = (node, e) => {
    const start = { x: node.x, y: node.y };
    onMouseDownMove(e, start, (dx, dy) => onChange({ x: start.x + dx, y: start.y + dy }));
  };

  const startResize = (node, e) => {
    const start = { w: node.w, h: node.h };
    onMouseDownMove(e, start, (dx, dy) => onChange({ w: Math.max(120, start.w + dx), h: Math.max(80, start.h + dy) }));
  };

  return { startDrag, startResize };
}

    const defaults = {
      Title: { props: { size: 24, weight: 800 }, content: 'New Title' },
      Paragraph: { props: { size: 14 }, content: 'New paragraph...' },
      Button: { props: { variant: 'filled', size: 'M' }, content: 'Click me' },
      Input: { props: { label: 'Label', radius: 8, placeholder: 'Enter...' }, content: '' },
      Card: { props: { title: 'Card title', subtitle: 'Subtitle', radius: 12 }, content: 'Card body content.' },
      Hero: { props: { heading: 'Hero title', sub: 'Catchy subheading', cta: 'Get started' }, content: '' },
    }[type] || {};
    setDoc(d => ({
      ...d,
      frames: d.frames.map(f => f.id === frameId ? { ...f, nodes: [...f.nodes, { ...base, ...defaults }] } : f)
    }));
  };

  const updateNode = (frameId, nodeId, patch) => {
    setDoc(d => ({
      ...d,
      frames: d.frames.map(f =>
        f.id === frameId ? { ...f, nodes: f.nodes.map(n => n.id === nodeId ? { ...n, ...patch } : n) } : f
      )
    }));
  };

  const removeNode = (frameId, nodeId) => {
    setDoc(d => ({
      ...d,
      frames: d.frames.map(f =>
        f.id === frameId ? { ...f, nodes: f.nodes.filter(n => n.id !== nodeId) } : f
      )
    }));
  };

  const setSelection = (sel) => setDoc(d => ({ ...d, selection: sel }));

  const serialize = () => JSON.stringify(doc, null, 2);
  const load = (json) => setDoc(JSON.parse(json));

  return { doc, addFrame, removeFrame, addNode, updateFrame, updateFrameStyle, updateNode, removeNode, setSelection, serialize, load };
}

function Frame({ frame, onMove, onResize, onSelect, onAddNode, docApi }) {
  const { startDrag, startResize } = useDragResize(patch => onMove(patch));
  const ref = useRef(null);
  const [menu, setMenu] = useState(null);

  const cssVars = useMemo(() => ({
    '--padding': `${frame.style?.padding ?? 12}px`,
    '--gap': `${frame.style?.gap ?? 8}px`,
    '--radius': `${frame.style?.radius ?? 12}px`,
    '--shadow': frame.style?.shadow ?? 'none',
  }), [frame.style]);

  return (
    <div
      className="frame"
      ref={ref}
      style={{ left: frame.x, top: frame.y, width: frame.w, height: frame.h, ...cssVars }}
      onMouseDown={(e) => {
        if (e.target.classList.contains('resize-handle')) return;
        onSelect({ frameId: frame.id, nodeId: null });
        startDrag(frame, e);
      }}
    >
      {/* plus button */}
      <button className="plus" onClick={(e) => {
        e.stopPropagation();
        const rect = e.currentTarget.getBoundingClientRect();
        setMenu({ x: rect.left, y: rect.bottom + 6 });
      }}>+ Add</button>
      {menu && <InsertMenu x={menu.x} y={menu.y} onPick={(type) => onAddNode(type)} onClose={() => setMenu(null)} />}

      {/* nodes */}
      <div style={{ overflow: 'auto' }}>
        {frame.nodes.map(node => {
          const C = Components[node.type];
          if (!C) return <div key={node.id}>Unknown: {node.type}</div>;
          return (
            <div key={node.id} onMouseDown={(e) => { e.stopPropagation(); onSelect({ frameId: frame.id, nodeId: node.id }); }}>
              <C
                node={node}
                onProp={(k, v) => docApi.updateNode(frame.id, node.id, { props: { ...node.props, [k]: v } })}
                onContent={(val) => docApi.updateNode(frame.id, node.id, { content: val })}
              />
            </div>
          );
        })}
      </div>

      <div className="resize-handle" onMouseDown={(e) => startResize(frame, e)} />
    </div>
  );
}

import { useEffect } from 'react';

export function useHotkeys(bindings) {
  useEffect(() => {
    const onKeyDown = (e) => {
      const combo = [
        e.ctrlKey || e.metaKey ? 'Mod' : null,
        e.shiftKey ? 'Shift' : null,
        e.altKey ? 'Alt' : null,
        e.key.length === 1 ? e.key.toUpperCase() : e.key,
      ].filter(Boolean).join('+');

      if (bindings[combo]) {
        const shouldPrevent = bindings[combo](e) !== false;
        if (shouldPrevent) {
          e.preventDefault();
          e.stopPropagation();
        }
      }
    };
    window.addEventListener('keydown', onKeyDown, { capture: true });
    return () => window.removeEventListener('keydown', onKeyDown, { capture: true });
  }, [bindings]);
}

function SortableNode({ frameId, node, onSelect, updateNode }) {
  const { attributes, listeners, setNodeRef, transform, transition } = useSortable({ id: node.id });
  const style = {
    transform: CSS.Transform.toString(transform),
    transition,
  };
  const C = Components[node.type];

  // State preview controls
  const [uiState, setUiState] = useState('default'); // default | hover | focus | active
  const wrapStyle = useMemo(() => {
    const common = { borderRadius: 10, border: '1px solid transparent' };
    const border = node.props.border || { width: 0, color: '#2b3856' };
    const shadow = node.props.shadow || 'none';
    if (uiState === 'hover') return { ...common, border: `${border.width || 0}px solid ${border.color || '#2b3856'}`, boxShadow: shadow, filter: 'brightness(1.02)' };
    if (uiState === 'focus') return { ...common, border: `1px solid #9bd0ff`, boxShadow: '0 0 0 3px rgba(155,208,255,.25)' };
    if (uiState === 'active') return { ...common, transform: 'scale(0.995)' };
    return common;
  }, [uiState, node.props.border, node.props.shadow]);

  return (
    <div ref={setNodeRef} style={style} onMouseDown={(e) => { e.stopPropagation(); onSelect({ frameId, nodeId: node.id }); }}>
      <div className="flex items-center gap-2">
        <span className="node-grip select-none" {...attributes} {...listeners}>⋮⋮</span>
        <div className="flex-1">
          <div className="rounded-lg" style={wrapStyle}
            onMouseEnter={() => setUiState('hover')}
            onMouseLeave={() => setUiState('default')}
            onMouseDown={() => setUiState('active')}
            onMouseUp={() => setUiState('hover')}
          >
            {C ? (
              <C
                node={node}
                onProp={(k, v) => updateNode(frameId, node.id, { props: { ...node.props, [k]: v } })}
                onContent={(val) => updateNode(frameId, node.id, { content: val })}
              />
            ) : (
              <div className="component">Unknown: {node.type}</div>
            )}
          </div>
        </div>
      </div>
    </div>
  );
}

  } = useDoc();

    if (!el) return;

      if (e.button !== 0) return;
      if ((e.target).closest('.plus') || (e.target).closest('.re-resizable')) return;

      if (!start) return;

      let ny = start.oy + (e.clientY - start.y);

      // snap logic

      }

        if (Math.abs((ny + frame.h / 2) - g.pos) < (guides.eps || 6)) { snapY = g.pos - frame.h / 2; }
        if (Math.abs((ny + frame.h) - g.pos) < (guides.eps || 6)) { snapY = g.pos - frame.h; }

      setDragging(false);
      setGuides({ v: [], h: [], snap: null });
      window.removeEventListener('mousemove', move);
      window.removeEventListener('mouseup', up);
    };
    el.addEventListener('mousedown', down);

export function exportFrameToReactTailwind(frame) {
  const esc = (s) => (s || '').replace(/`/g, '\\`');

  const renderNode = (n) => {

      case 'Paragraph':
        return `<p className="text-[${n.props.size || 14}px] text-slate-300">${esc(n.content || 'Paragraph')}</p>`;
      case 'Button': {
        const base = `px-3 py-2 rounded-${(n.props.radius ?? 8) >= 12 ? 'xl' : 'lg'} font-semibold`;

      }
      case 'Hero':

  <div className="grid grid-cols-[1.2fr_1fr] gap-4">

// Context-aware quick suggestions by analyzing nodes in frame
export function getContextualSuggestions(nodes) {
  const types = new Set(nodes.map(n => n.type));
  const suggestions = [];

  if (types.has('Title') && !types.has('Paragraph')) suggestions.push('Paragraph');
  if (types.has('Paragraph') && !types.has('Button')) suggestions.push('Button');
  if (!types.has('Input') && (types.has('Title') || types.has('Paragraph'))) suggestions.push('Input');
  if (!types.has('Hero') && (types.size < 3)) suggestions.push('Hero');
  if (!types.has('Card')) suggestions.push('Card');

  // Always include quick patterns
  suggestions.push('Title', 'Paragraph', 'Button');
  // Deduplicate with stable order
  const seen = new Set();
  return suggestions.filter(s => (seen.has(s) ? false : (seen.add(s), true)));
}

// WCAG contrast calculation (approx)
function luminance(hex) {
  const c = hex.replace('#','');
  const r = parseInt(c.substring(0,2),16) / 255;
  const g = parseInt(c.substring(2,4),16) / 255;
  const b = parseInt(c.substring(4,6),16) / 255;
  const srgb = [r,g,b].map(v => v <= 0.03928 ? v/12.92 : Math.pow((v+0.055)/1.055, 2.4));
  return 0.2126*srgb[0] + 0.7152*srgb[1] + 0.0722*srgb[2];
}
export function contrastRatio(fg, bg) {
  try {
    const L1 = luminance(fg);
    const L2 = luminance(bg);
    const lighter = Math.max(L1, L2);
    const darker = Math.min(L1, L2);
    return (lighter + 0.05) / (darker + 0.05);
  } catch { return null; }
}
export function passLevel(ratio, sizePx=14, bold=false) {
  if (ratio == null) return 'N/A';
  // AA: 4.5 for normal, 3.0 for large (>=18pt or 14pt bold ~ 24px or 18.66px bold)
  const large = sizePx >= 24 || (bold && sizePx >= 18.66);
  if (ratio >= 7) return 'AAA';
  if (ratio >= 4.5 && !large) return 'AA';
  if (ratio >= 3 && large) return 'AA Large';
  return 'Fail';
}

export function withHistory(store, opts = { limit: 50 }) {
  let history = [];
  let idx = -1;
  const save = (state) => {
    // prune forward
    if (idx < history.length - 1) history = history.slice(0, idx + 1);
    history.push(state);
    if (history.length > opts.limit) history.shift();
    idx = history.length - 1;
  };
  const getState = () => store.getState();
  const setState = (partial) => {
    const prev = getState();
    store.setState(partial);
    const next = getState();
    save(JSON.stringify({ doc: next.doc, theme: next.theme }));
  };
  const undo = () => {
    if (idx <= 0) return;
    idx -= 1;
    const snap = JSON.parse(history[idx]);
    store.setState({ doc: snap.doc, theme: snap.theme });
  };
  const redo = () => {
    if (idx >= history.length - 1) return;
    idx += 1;
    const snap = JSON.parse(history[idx]);
    store.setState({ doc: snap.doc, theme: snap.theme });
  };
  const bootstrap = () => save(JSON.stringify({ doc: getState().doc, theme: getState().theme }));
  return { setState, undo, redo, bootstrap, _debugHistory: () => ({ history, idx }) };
}

import { useSettings } from './settings.js';
import { useDoc } from './store.js';

export async function callLLM(userText, opts = { includeContext: true }) {
  const { ai } = useSettings.getState();
  if (!ai.apiKey) throw new Error('Missing API key in Settings');

  const messages = [];
  messages.push({ role: 'system', content: ai.systemPrompt || '' });

  if (opts.includeContext && ai.sendContext) {
    // Minimal context: selected frame and nodes
    const { frame, nodes } = useDoc.getState().getSelected();
    const context = {
      frame: frame ? { id: frame.id, name: frame.name, w: frame.w, h: frame.h, grid: frame.grid, breakpoints: frame.breakpoints } : null,
      nodes: nodes?.map(n => ({ id: n.id, type: n.type, content: n.content || '', props: n.props || {} })) || [],
      tokens: useDoc.getState().theme || {},
    };
    messages.push({ role: 'system', content: `Context:\n${JSON.stringify(context).slice(0, 8000)}` });
  }

  messages.push({ role: 'user', content: userText });

  const body = {
    model: ai.model,
    temperature: ai.temperature ?? 0.2,
    messages,
    response_format: { type: 'json_object' },
  };

  const headers = {
    'Content-Type': 'application/json',
    Authorization: ai.provider === 'openrouter' ? `Bearer ${ai.apiKey}` : `Bearer ${ai.apiKey}`,
    ...(ai.provider === 'openrouter' ? {
      'HTTP-Referer': 'https://your-app.local', // optional
      'X-Title': 'FrameForge',
    } : {}),
  };

  const res = await fetch(ai.endpoint, {
    method: 'POST',
    headers,
    body: JSON.stringify(body),
  });

  if (!res.ok) {
    const text = await res.text().catch(()=> '');
    throw new Error(`LLM error ${res.status}: ${text}`);
  }

  const data = await res.json();
  // OpenRouter and OpenAI-compatible share structure
  const content = data.choices?.[0]?.message?.content || '';
  let json = null;
  try { json = JSON.parse(content); } catch (e) {
    // Try to find JSON substring
    const match = content.match(/\{[\s\S]*\}$/);
    if (match) { try { json = JSON.parse(match[0]); } catch {} }
  }
  return json;
}

function applyAIActions(result) {
  const actions = Array.isArray(result) ? result : (result?.actions || []);
  if (!actions || !actions.length) return;

  const S = useDoc.getState();
  actions.forEach(act => {
    if (act.action === 'insert' && act.frameId && Array.isArray(act.nodes)) {
      act.nodes.forEach(n => {
        S.addNode(act.frameId, n.type);
        const f = useDoc.getState().doc.frames.find(fr => fr.id === act.frameId);
        const lastId = f?.nodes?.at(-1)?.id;
        if (lastId) S.updateNode(act.frameId, lastId, { props: n.props || {}, content: n.content || '' });
      });
    } else if (act.action === 'update' && act.frameId && Array.isArray(act.targets)) {

import { useSettings } from './settings.js';

let discovered = null;

export async function mcpDiscover() {
  const { mcp } = useSettings.getState();
  if (!mcp?.endpoint) return { tools: [] };
  const res = await fetch(`${mcp.endpoint.replace(/\/$/,'')}/tools`, {
    headers: mcp.apiKey ? { Authorization: `Bearer ${mcp.apiKey}` } : {},
  });
  if (!res.ok) throw new Error(`MCP discovery failed: ${res.status}`);
  const data = await res.json();
  discovered = data;
  return data;
}

export async function callMCP(name, args) {
  const { mcp } = useSettings.getState();
  if (!mcp?.endpoint) throw new Error('No MCP endpoint configured');
  const res = await fetch(`${mcp.endpoint.replace(/\/$/,'')}/call`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      ...(mcp.apiKey ? { Authorization: `Bearer ${mcp.apiKey}` } : {}),
    },
    body: JSON.stringify({ name, arguments: args || {} }),
  });
  if (!res.ok) {
    const t = await res.text().catch(()=> '');
    throw new Error(`MCP call error ${res.status}: ${t}`);
  }
  return await res.json();
}

export function getDiscoveredTools() {
  return discovered?.tools || [];
}

import { useSettings } from './settings.js';
import { useDoc } from './store.js';

export async function callLLM(userText, opts = { includeContext: true }) {
  const { ai } = useSettings.getState();
  if (!ai.apiKey) throw new Error('Missing API key in Settings');

  const messages = [];
  messages.push({ role: 'system', content: ai.systemPrompt || '' });

  if (opts.includeContext && ai.sendContext) {
    const { frame, nodes } = useDoc.getState().getSelected();
    const context = {
      frame: frame ? { id: frame.id, name: frame.name, w: frame.w, h: frame.h, grid: frame.grid, breakpoints: frame.breakpoints } : null,
      nodes: nodes?.map(n => ({ id: n.id, type: n.type, content: n.content || '', props: n.props || {} })) || [],
      tokens: useDoc.getState().theme || {},
    };
    messages.push({ role: 'system', content: `Context:\n${JSON.stringify(context).slice(0, 8000)}` });
  }

  messages.push({ role: 'user', content: userText });

  const body = {
    model: ai.model,
    temperature: ai.temperature ?? 0.2,
    messages,
    // Encourage JSON
    response_format: { type: 'json_object' },
  };

  const headers = {
    'Content-Type': 'application/json',
    Authorization: `Bearer ${ai.apiKey}`,
    ...(ai.provider === 'openrouter' ? {
      'HTTP-Referer': 'https://frameforge.local',
      'X-Title': 'FrameForge',
    } : {}),
  };

  const res = await fetch(ai.endpoint, {
    method: 'POST',
    headers,
    body: JSON.stringify(body),
  });

  if (!res.ok) {
    const text = await res.text().catch(()=> '');
    throw new Error(`LLM error ${res.status}: ${text}`);
  }

  const data = await res.json();
  const content = data.choices?.[0]?.message?.content || '';
  // Try parse JSON
  try {
    return JSON.parse(content);
  } catch (e) {
    // Fallback: extract last JSON object in text
    const match = content.match(/\{[\s\S]*\}$/);
    if (match) {
      try { return JSON.parse(match[0]); } catch {}
    }
    throw new Error('AI did not return valid JSON. Try again.');
  }
}

import { useSettings } from './settings.js';

let discovered = null;

export async function mcpDiscover() {
  const { mcp } = useSettings.getState();
  if (!mcp?.endpoint) return { tools: [] };
  const res = await fetch(`${mcp.endpoint.replace(/\/$/,'')}/tools`, {
    headers: mcp.apiKey ? { Authorization: `Bearer ${mcp.apiKey}` } : {},
  });
  if (!res.ok) throw new Error(`MCP discovery failed: ${res.status}`);
  const data = await res.json();
  discovered = data;
  return data;
}

export async function callMCP(name, args) {
  const { mcp } = useSettings.getState();
  if (!mcp?.endpoint) throw new Error('No MCP endpoint configured');
  const res = await fetch(`${mcp.endpoint.replace(/\/$/,'')}/call`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      ...(mcp.apiKey ? { Authorization: `Bearer ${mcp.apiKey}` } : {}),
    },
    body: JSON.stringify({ name, arguments: args || {} }),
  });
  if (!res.ok) {
    const t = await res.text().catch(()=> '');
    throw new Error(`MCP call error ${res.status}: ${t}`);
  }
  return await res.json();
}

export function getDiscoveredTools() {
  return discovered?.tools || [];
}

function applyAIActions(result) {
  const actions = Array.isArray(result) ? result : (result?.actions || []);
  if (!actions || !actions.length) return;

  const S = useDoc.getState();
  actions.forEach(act => {
    if (act.action === 'insert' && act.frameId && Array.isArray(act.nodes)) {
      act.nodes.forEach(n => {
        S.addNode(act.frameId, n.type);
        const f = useDoc.getState().doc.frames.find(fr => fr.id === act.frameId);
        const lastId = f?.nodes?.at(-1)?.id;
        if (lastId) S.updateNode(act.frameId, lastId, { props: n.props || {}, content: n.content || '' });
      });
    } else if (act.action === 'update' && act.frameId && Array.isArray(act.targets)) {
      act.targets.forEach(id => S.updateNode(act.frameId, id, act.props || {}));
    } else if (act.action === 'critique' && act.frameId) {
      const notes = act.notes || [];
      const patches = act.suggestedPatches || [];
      S.updateFrame(act.frameId, { aiNotes: notes, aiPatches: patches });
    }
  });
}

async function handleAIResult(result) {
  if (!result) return;
  // Tool call?
  if (result.tool?.name) {
    const { mcp } = useSettings.getState();
    const name = result.tool.name;
    if (!mcp.allowedTools.includes(name)) {
      throw new Error(`Tool not allowed: ${name}`);
    }
    const out = await dispatchTool(name, result.tool.arguments || {});
    if (out?.code && out?.filename) {
      const blob = new Blob([out.code], { type: 'text/plain' });
      const a = document.createElement('a');
      a.href = URL.createObjectURL(blob);
      a.download = out.filename;
      a.click();
    }
    if (out?.notes) {
      const S = useDoc.getState();
      const fr = S.getSelected().frame;
      if (fr) S.updateFrame(fr.id, { aiNotes: out.notes });
    }
    return;
  }
  // Else actions
  applyAIActions(result.actions || result);
}

function AIAssistant() {
  const [input, setInput] = React.useState('');
  const [busy, setBusy] = React.useState(false);
  const { frame } = useDoc((s) => s.getSelected());
  const execute = async (prompt) => {
    if (!prompt) return;
    setBusy(true);
    try {
      const result = await callLLM(prompt, { includeContext: true });
      await handleAIResult(result);
    } catch (e) {
      alert(e.message);
    } finally {
      setBusy(false);
    }
  };
  const quicks = [
    { label: 'Signup form', prompt: 'Insert a signup form with email, password, and a primary submit button. Use semantic components.' },
    { label: 'Hero section', prompt: 'Insert a landing hero with heading, subheading, primary and secondary buttons, and an image placeholder.' },
    { label: 'Critique', prompt: 'Critique the current frame for accessibility and hierarchy. Suggest patches as JSON.' },
    { label: 'Mobile-first', prompt: 'Refine the selected frame to be mobile-first. Reduce spacing, ensure 16px body text, full-width primary button.' },
    { label: 'Tighten spacing', prompt: 'Reduce vertical rhythm by ~20% across selected nodes. Keep contrast AA.' },
    { label: 'Export MUI (tool)', prompt: 'Call tool export.mui for the selected frame.' },
  ];
  return (
    <div className="panel ai">
      <div className="row" style={{ gap: 6, flexWrap: 'wrap' }}>
        {quicks.map(q => (

            if (q.label.includes('(tool)')) {
              // Example tool shortcut
              await handleAIResult({ tool: { name: 'export.mui', arguments: { frameId: frame?.id } } });
            } else {
              await execute(q.prompt);
            }
          }}>{q.label}</button>
        ))}
      </div>
      <div style={{ marginTop: 8 }}>
        <textarea rows={4} placeholder={frame ? 'Ask the design copilot…' : 'Select a frame to use AI'} value={input} onChange={(e)=>setInput(e.target.value)} />
      </div>
      <div className="row">
        <button disabled={busy || !frame || !input.trim()} onClick={()=>execute(input.trim())}>{busy ? 'Thinking…' : 'Run'}</button>
      </div>
      {frame?.aiNotes?.length ? (
        <div className="section">
          <strong>AI Critique</strong>
          <ul>
            {frame.aiNotes.map((n, i) => <li key={i} style={{ fontSize: 12 }}>{n}</li>)}
          </ul>
          {Array.isArray(frame.aiPatches) && frame.aiPatches.length > 0 && (
            <button onClick={()=>{
              const S = useDoc.getState();
              frame.aiPatches.forEach(p => p.targets.forEach(id => S.updateNode(frame.id, id, p.props || {})));
              S.updateFrame(frame.id, { aiPatches: [], aiNotes: [] });
            }}>Apply Suggested Fixes</button>
          )}
        </div>
      ) : null}
    </div>
  );
}
